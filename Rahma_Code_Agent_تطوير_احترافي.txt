================================================================================
                القسم 1: الخلاصة التنفيذية
================================================================================

تطبيقك "Rahma Code Agent" يملك الهيكل الصحيح (Electron + React + TypeScript +
SQLite + MCP + subagents + approval queue + DPAPI key storage) لكنه يُعاني من
تراكم اختصارات في التصميم تُفقده فعاليته في المهام الحقيقية.

أبرز المشاكل المُكتشفة:

  1. لا يستخدم ripgrep - البحث بـ RegExp بطيء جداً (100x أبطأ)
  2. يستخدم tokenizer خطأ لكل النماذج (gpt-tokenizer لـ Kimi/Qwen/DeepSeek/GLM)
  3. حد 30 خطوة بلا استمرار تلقائي أو كشف للحلقات العالقة
  4. System Prompt عربي بحت بلا أمثلة عملية (few-shot)
  5. أداتا تعديل متنافستان (edit_file + patch_file) مع مطابقة ضعيفة
  6. PowerShell بلا sandbox وفلتر خطر قابل للتجاوز
  7. الواجهة تعرض JSON خام - مكتبة xterm موجودة لكن غير مستخدمة
  8. وكلاء فرعيون للقراءة فقط (لا يستطيعون الكتابة)
  9. بحث الرموز بـ Regex heuristics بدل AST/LSP
 10. لا يوجد Plan mode حقيقي - todos لا تقود الحلقة

الخبر الجيد: كل هذه المشاكل ميكانيكية وقابلة للإصلاح في 6-8 أسابيع.
الخبر السيئ: بدون إصلاحها، التطبيق سيبقى "يبدو مكتمل لكنه يفشل في 5-15 خطوة".

```ts
const child = spawn(executable, ['-NoLogo', '-NoProfile', '-NonInteractive',
  '-Command', command], { cwd, windowsHide: true, env: safeEnvironment() })

function isCriticalCommand(command: string): boolean {
  return /(?:Remove-Item\s+.*(?:-Recurse|-Force)|Format-Volume|Clear-Disk|
    Stop-Computer|Restart-Computer|Set-MpPreference|reg(?:\.exe)?\s+delete|
    diskpart|bcdedit|cipher\s*\/w|taskkill\s*.*\/f)/i.test(command)
}
```

الثغرات الأمنية:
1. -Command يدمج كل شيء في script block واحد (مشاكل quoting)
2. الـ regex يفوّت:
   - del, rd /s, erase (أسماء مستعارة لـ Remove-Item)
   - cmd /c format C:
   - Invoke-WebRequest + Invoke-Expression (تحميل وتنفيذ)
   - New-ItemProperty / Set-ItemProperty على الـ registry
   - Stop-Process (لا حاجة لـ taskkill)
   - net user /delete, net localgroup administrators /add
   - أوامر WMI: ([wmiclass]"Win32_Process").Create("...")
   - Start-Process خارج مساحة العمل
3. لا AppContainer، لا Job Object، لا restricted token
4. لا ConstrainedLanguage mode
5. لا syscall filtering
6. cwd يضعه النموذج بنفسه

في وضع Full Access (زر بارز في الواجهة) → النموذج يمكنه تدمير النظام.

```python
while not task_complete and steps < MAX_STEPS:
    # 1. كشف الحلقات العالقة
    if is_stuck_loop(history, last_n=10):
        inject_system_message("لقد جربت نفس الشيء 3 مرات. غيّر نهجك.")

    # 2. تذكير بالـ todo الحالي
    current_todo = get_next_todo()
    inject_context(f"الخطوة الحالية: {current_todo}")

    # 3. اختيار النموذج المناسب للمهمة
    model = route_model(task_type)

    # 4. تنفيذ متوازي للقراءات
    if all_read_tools(tool_calls):
        results = await asyncio.gather(*[execute(tc) for tc in tool_calls])

    # 5. retry budget لكل أداة
    for call in tool_calls:
        result = await execute_with_retry(call, max_retries=2)
        if result.failed:
            track_failure(call.tool_name)
            if consecutive_failures(call.tool_name) >= 3:
                pause_and_ask_user()

    # 6. عند الاقتراب من الحد، استمر تلقائياً
    if steps == MAX_STEPS - 3 and todos_remaining > 0:
        inject_user_message("أكمل من حيث توقفت")
        MAX_STEPS += 30
```

// src/main/search-ripgrep.ts
import { spawn } from 'node:child_process'
import path from 'node:path'

export interface RipgrepMatch {
  type: 'match' | 'summary'
  data: {
    path?: { text: string }
    line_number?: number
    absolute_offset?: number
    submatches?: Array<{
      match: { text: string }
      start: number
      end: number
    }>
    bytes?: string
    elapsed_total?: { secs: number, nanos: number }
    stats?: {
      elapsed: { secs: number, nanos: number }
      matches: number
      matches_with_lines: number
      bytes_searched: number
      bytes_printed: number
      bytes_searched_binary: number
    }
  }
}

export async function searchWithRipgrep(
  rootPath: string,
  pattern: string,
  options: {
    glob?: string[]
    ignoreGlob?: string[]
    fixed?: boolean
    maxResults?: number
    signal?: AbortSignal
  } = {}
): Promise<RipgrepMatch[]> {
  const args = [
    '--json',
    '--max-count', String(options.maxResults ?? 500),
    '--no-config',
    '--no-ignore-vcs',  // سنتحكم يدوياً
    '--hidden',         // ابحث في الملفات المخفية لكن احترم .gitignore
  ]

  if (options.fixed) args.push('--fixed-strings')
  if (options.glob) {
    for (const g of options.glob) args.push('--glob', g)
  }
  if (options.ignoreGlob) {
    for (const g of options.ignoreGlob) args.push('--glob', `!${g}`)
  }

  args.push(pattern, rootPath)

  return new Promise((resolve, reject) => {
    const rg = spawn('rg', args, { windowsHide: true })
    const results: RipgrepMatch[] = []
    let stderr = ''

    const timer = setTimeout(() => {
      rg.kill('SIGTERM')
      reject(new Error('ripgrep timeout'))
    }, 30_000)

    rg.stdout.on('data', (chunk) => {
      const lines = chunk.toString().split('\n').filter(Boolean)
      for (const line of lines) {
        try {
          results.push(JSON.parse(line))
        } catch { /* skip invalid */ }
      }
    })

    rg.stderr.on('data', (chunk) => { stderr += chunk.toString() })

    rg.on('close', (code) => {
      clearTimeout(timer)
      if (code === 0 || code === 1) {  // 0 = matches, 1 = no matches
        resolve(results)
      } else {
        reject(new Error(`ripgrep failed (${code}): ${stderr}`))
      }
    })

    if (options.signal) {
      options.signal.addEventListener('abort', () => {
        rg.kill('SIGTERM')
        clearTimeout(timer)
      })
    }
  })
}

// src/main/edit-block.ts
export interface EditBlock {
  filePath: string
  searchBlock: string
  replaceBlock: string
}

const SEARCH_REPLACE_REGEX = /<<<<<<< SEARCH\n([\s\S]*?)\n=======\n([\s\S]*?)\n>>>>>>> REPLACE/g

export function parseEditBlocks(content: string): EditBlock[] {
  const blocks: EditBlock[] = []
  const lines = content.split('\n')

  let i = 0
  while (i < lines.length) {
    const line = lines[i]

    // ابحث عن "```language path"
    const fenceMatch = line.match(/^```(\w+)?\s+(.+?)$/)
    if (fenceMatch) {
      const filePath = fenceMatch[2]!.trim()
      i++

      // اجمع محتوى الكود
      const codeLines: string[] = []
      while (i < lines.length && lines[i] !== '```') {
        codeLines.push(lines[i]!)
        i++
      }
      i++  // skip closing ```

      const code = codeLines.join('\n')
      const matches = [...code.matchAll(SEARCH_REPLACE_REGEX)]

      for (const match of matches) {
        blocks.push({
          filePath,
          searchBlock: match[1]!,
          replaceBlock: match[2]!,
        })
      }
    } else {
      i++
    }
  }

  return blocks
}

export function applySearchReplace(
  content: string,
  search: string,
  replace: string
): { result: string; matchQuality: number; matched: boolean } {
  // 1. محاولة مطابقة تامة
  const exactIndex = content.indexOf(search)
  if (exactIndex !== -1) {
    return {
      result: content.slice(0, exactIndex) + replace + content.slice(exactIndex + search.length),
      matchQuality: 1.0,
      matched: true,
    }
  }

  // 2. مطابقة مع تطبيع نهايات الأسطر فقط
  const normalizedContent = content.replace(/\r\n/g, '\n').replace(/\r/g, '\n')
  const normalizedSearch = search.replace(/\r\n/g, '\n').replace(/\r/g, '\n')
  const normIndex = normalizedContent.indexOf(normalizedSearch)
  if (normIndex !== -1) {
    return {
      result: normalizedContent.slice(0, normIndex) + replace + normalizedContent.slice(normIndex + normalizedSearch.length),
      matchQuality: 0.95,
      matched: true,
    }
  }

  // 3. fuzzy matching (تطبيع whitespace)
  const fuzzyContent = normalizedContent.replace(/[ \t]+/g, ' ').replace(/ +$/gm, '')
  const fuzzySearch = normalizedSearch.replace(/[ \t]+/g, ' ').replace(/ +$/gm, '')
  const fuzzyIndex = fuzzyContent.indexOf(fuzzySearch)
  if (fuzzyIndex !== -1) {
    // طبّق على الـ normalized content
    return {
      result: normalizedContent.slice(0, fuzzyIndex) + replace + normalizedContent.slice(fuzzyIndex + fuzzySearch.length),
      matchQuality: 0.7,
      matched: true,
    }
  }

  return { result: content, matchQuality: 0, matched: false }
}

// الاستخدام:
const blocks = parseEditBlocks(modelResponse)
for (const block of blocks) {
  const content = await fs.readFile(block.filePath, 'utf8')
  const { result, matchQuality, matched } = applySearchReplace(
    content,
    block.searchBlock,
    block.replaceBlock
  )
  if (!matched) {
    throw new Error(`لم يتم العثور على SEARCH block في ${block.filePath}`)
  }
  if (matchQuality < 0.9) {
    console.warn(`تحذير: جودة التطابق ${matchQuality} في ${block.filePath}`)
  }
  await fs.writeFile(block.filePath, result)
}

// src/main/stuck-detector.ts
interface ToolCallRecord {
  toolName: string
  inputHash: string
  timestamp: number
  success: boolean
}

export class StuckLoopDetector {
  private history: ToolCallRecord[] = []
  private readonly windowSize = 10
  private readonly repeatThreshold = 3
  private readonly consecutiveFailureThreshold = 5

  record(toolName: string, input: unknown, success: boolean): void {
    const inputHash = this.hash(JSON.stringify(input))
    this.history.push({
      toolName,
      inputHash,
      timestamp: Date.now(),
      success,
    })

    // احتفظ بآخر 50 سجل فقط
    if (this.history.length > 50) {
      this.history.shift()
    }
  }

  isStuckLoop(): boolean {
    if (this.history.length < this.repeatThreshold) return false
    const recent = this.history.slice(-this.windowSize)
    const last = recent[recent.length - 1]!

    // عدّ التكرارات لنفس (tool, input) في النافذة
    const repeats = recent.filter(
      r => r.toolName === last.toolName && r.inputHash === last.inputHash
    ).length

    return repeats >= this.repeatThreshold
  }

  hasConsecutiveFailures(toolName?: string): boolean {
    const recent = this.history.slice(-this.consecutiveFailureThreshold)
    if (recent.length < this.consecutiveFailureThreshold) return false

    return recent.every(r => !r.success && (toolName ? r.toolName === toolName : true))
  }

  getStuckLoopMessage(): string {
    const last = this.history[this.history.length - 1]
    if (!last) return ''
    return `لقد جربت ${last.toolName} بنفس المدخلات ${this.repeatThreshold} مرات في آخر ${this.windowSize} خطوات. ` +
           `هذا نهج غير فعال. جرّب:\n` +
           `- إعادة قراءة الملف للتأكد من حالته الحالية\n` +
           `- استخدام search-and-replace بدل التعديل المباشر\n` +
           `- تقسيم المهمة لخطوات أصغر\n` +
           `- طلب توضيح من المستخدم`
  }

  getConsecutiveFailureMessage(): string {
    return `فشلت 5 محاولات متتالية. توقفت لتجنب إهدار الموارد. ` +
           `راجع الأخطاء أعلاه وقرر: هل تريد المتابعة بنهج مختلف؟`
  }

  private hash(s: string): string {
    let h = 0
    for (let i = 0; i < s.length; i++) {
      h = ((h << 5) - h + s.charCodeAt(i)) | 0
    }
    return String(h)
  }
}

// الاستخدام في agent loop:
const detector = new StuckLoopDetector()

for (let step = 0; step < MAX_STEPS; step++) {
  // ... تنفيذ tool call ...
  detector.record(call.name, call.input, result.success)

  if (detector.isStuckLoop()) {
    messages.push({

    })
  }

  if (detector.hasConsecutiveFailures()) {
    await pauseAndAskUser(detector.getConsecutiveFailureMessage())

  }

  // auto-continue عند الاقتراب من الحد

    if (todosRemaining > 0) {
      messages.push({
        role: 'user',
        content: `أكمل من حيث توقفت. تبقى ${todosRemaining} مهام.`,
      })
      MAX_STEPS += 30
    }
  }
}

// src/main/provider.ts (مُحدّث)
function toAnthropicBody(config, messages, tools, maxOutput): Record<string, unknown> {
  const systemBlocks = buildSystemBlocks(messages)
  const toolDefs = buildToolDefs(tools)

  // 1. cache breakpoint على system prompt
  if (systemBlocks.length) {
    systemBlocks.at(-1)!.cache_control = { type: 'ephemeral' }
  }

  // 2. cache breakpoint على tool definitions
  if (toolDefs.length) {
    toolDefs.at(-1)!.cache_control = { type: 'ephemeral' }
  }

  // 3. cache breakpoint على منتصف المحادثة
  const msgBlocks = messages.filter(m => m.role !== 'system')
  if (msgBlocks.length >= 4) {
    const midIdx = Math.floor(msgBlocks.length / 2)
    const midMsg = msgBlocks[midIdx]
    if (Array.isArray(midMsg.content)) {
      midMsg.content.at(-1).cache_control = { type: 'ephemeral' }
    } else if (typeof midMsg.content === 'string') {
      midMsg.content = [{ type: 'text', text: midMsg.content, cache_control: { type: 'ephemeral' } }]
    }
  }

  // 4. cache breakpoint على آخر رسالة
  const lastMsg = msgBlocks[msgBlocks.length - 1]
  if (lastMsg && Array.isArray(lastMsg.content)) {
    lastMsg.content.at(-1).cache_control = { type: 'ephemeral' }
  } else if (lastMsg && typeof lastMsg.content === 'string') {
    lastMsg.content = [{ type: 'text', text: lastMsg.content, cache_control: { type: 'ephemeral' } }]
  }

  return {
    model: config.model,
    system: systemBlocks,
    tools: toolDefs,
    messages: msgBlocks,
    max_tokens: maxOutput,
    temperature: 0,
  }
}

// src/main/powershell-sandbox.ts
const ALLOWED_COMMANDS = new Set([
  'npm', 'node', 'npx', 'tsc', 'git', 'cargo',
  'python', 'pytest', 'go', 'rustc', 'make',
  'dotnet', 'java', 'javac', 'mvn', 'gradle',
  'ls', 'dir', 'cat', 'echo', 'cd', 'pwd',
])

const BLOCKED_PATTERNS = [
  /\biex\b/i,                                    // Invoke-Expression
  /\biwr\b.*\|.*\biex\b/i,                       // download + execute
  /Invoke-WebRequest.*\|.*Invoke-Expression/i,
  /Start-Process(?![^\n]*-WorkingDirectory)/i,   // Start-Process outside cwd
  /\breg\b.*\bdelete\b/i,                        // registry delete
  /\breg\b.*\badd\b/i,                           // registry add (require approval)
  /Stop-Process/i,
  /\bnet\b\s+user/i,                             // user management
  /\bnet\b\s+localgroup/i,
  /\[wmiclass\]/i,                               // WMI process creation
  /Format-Volume/i,
  /Clear-Disk/i,
  /diskpart/i,
  /bcdedit/i,
  /cipher\s*\/w/i,
  /Set-MpPreference/i,                           // Windows Defender
  /Remove-Item\s+.*-Recurse.*-Force/i,
  /\bdel\s+\/[sS]/i,                             // del /s
  /\brd\s+\/[sS]/i,                              // rd /s
]

export function validatePowerShellCommand(command: string): {
  valid: boolean
  reason?: string
} {
  // 1. تحقق من الأنماط الخطرة
  for (const pattern of BLOCKED_PATTERNS) {
    if (pattern.test(command)) {
      return {
        valid: false,
        reason: `Blocked pattern detected: ${pattern.source}. ` +
                `If this is a legitimate operation, the user must approve it manually.`,
      }
    }
  }

  // 2. تحقق من أن الأمر الأول في الـ allowlist (للأوامر البسيطة)
  const firstWord = command.trim().split(/\s+/)[0]?.replace(/\.exe$/i, '')
  if (firstWord && !ALLOWED_COMMANDS.has(firstWord)) {
    return {
      valid: false,
      reason: `Command "${firstWord}" is not in the allowed list. ` +
              `Allowed: ${[...ALLOWED_COMMANDS].join(', ')}.`,
    }
  }

  return { valid: true }
}

// الاستخدام:
const validation = validatePowerShellCommand(command)
if (!validation.valid) {
  return {
    ok: false,
    error: validation.reason,
    suggestion: 'If you need this command, ask the user to approve it manually.',
  }
}

import { useEffect, useRef } from 'react'
import { Terminal } from '@xterm/xterm'
import { FitAddon } from '@xterm/addon-fit'
import '@xterm/xterm/css/xterm.css'

interface TerminalViewProps {
  output: string
  ansi?: boolean
}

export function TerminalView({ output, ansi = true }: TerminalViewProps) {
  const containerRef = useRef<HTMLDivElement>(null)
  const termRef = useRef<Terminal | null>(null)

  useEffect(() => {
    if (!containerRef.current) return

    const term = new Terminal({
      fontFamily: 'Consolas, "Courier New", monospace',
      fontSize: 13,
      theme: {
        background: '#1e1e1e',
        foreground: '#d4d4d4',
        cursor: '#d4d4d4',
      },
      scrollback: 5000,
      convertEol: true,
    })
    const fitAddon = new FitAddon()
    term.loadAddon(fitAddon)
    term.open(containerRef.current)
    fitAddon.fit()

    termRef.current = term

    const handleResize = () => fitAddon.fit()
    window.addEventListener('resize', handleResize)

    return () => {
      window.removeEventListener('resize', handleResize)
      term.dispose()
    }
  }, [])

  useEffect(() => {
    if (termRef.current && output) {
      termRef.current.write(output)
    }
  }, [output])

  return <div ref={containerRef} style={{ height: '300px', width: '100%' }} />
}
